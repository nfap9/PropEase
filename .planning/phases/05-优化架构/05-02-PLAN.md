1. 添加 AdminApiConfig 类型:

```typescript
export interface AdminApiConfig {
  baseURL: string;
  adminTokenKey?: string;  // 默认为 'admin_access_token'
  loginPath?: string;       // 默认为 '/login'
}
```

2. 创建 createAdminApiClient 函数:
```typescript
export function createAdminApiClient(config: AdminApiConfig): AxiosInstance {
  const tokenStorage = createLocalStorageTokenStorage(
    config.adminTokenKey ?? 'admin_access_token',
    'admin_refresh_token'  // admin 不使用 refresh token，设为默认值
  );

  return createApiClient({
    baseURL: config.baseURL,
    tokenStorage,
    refreshTokens: undefined,  // admin 不需要 refresh token
    onAuthFailure: () => {
      if (typeof window === 'undefined') return;
      if (!window.location.pathname.startsWith(config.loginPath ?? '/login')) {
        window.location.href = config.loginPath ?? '/login';
      }
    },
    suppressUnauthorizedError: true,  // admin 401 处理不显示错误
  });
}
```

3. 在文件底部添加导出:
```typescript
export { createAdminApiClient };
```

注意: adminApi 响应拦截器需要特殊处理 401（清除 admin_access_token 并跳转登录页），createApiClient 的 onAuthFailure 已经处理了这个逻辑。
