Create new UnifiedLayout component that:

1. Reads `isAdmin` from useAuth() context
2. Conditionally renders either AdminLayout (when isAdmin=true) or MainLayout (when isAdmin=false)
3. Passes children to the selected layout

```typescript
'use client';

import { useAuth } from '@/lib/auth/context';
import { AdminLayout } from './admin-layout';
import { MainLayout } from './main-layout';

interface UnifiedLayoutProps {
  children: React.ReactNode;
}

export function UnifiedLayout({ children }: UnifiedLayoutProps) {
  const { isAdmin } = useAuth();

  if (isAdmin) {
    return <AdminLayout>{children}</AdminLayout>;
  }

  return <MainLayout>{children}</MainLayout>;
}
```

This is the key routing decision component - it decides which layout to show based on auth state.
